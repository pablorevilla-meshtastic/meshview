// Base map tiles from the [site] map_tile_* options (OpenStreetMap by default). window.meshMapTiles = {{ map_tiles|tojson }}; window.meshTileLayer = (options = {}) => L.tileLayer(meshMapTiles.url, { maxZoom: meshMapTiles.max_zoom, attribution: meshMapTiles.attribution, ...options }); // Topographic tiles, offered only with the default tiles: a custom tile // server usually means a network without Internet access. window.meshTopoLayer = () => meshMapTiles.custom ? null : L.tileLayer("https://{s}.tile.opentopomap.org/{z}/{x}/{y}.png", { maxZoom: 17, attribution: 'Map data: © OpenStreetMap contributors, SRTM | Map style: © OpenTopoMap (CC-BY-SA)' });
